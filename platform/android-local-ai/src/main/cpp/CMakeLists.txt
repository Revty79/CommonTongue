cmake_minimum_required(VERSION 3.22)
project(common_tongue_local_ai LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized local inference" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Pinned prepared sources only" FORCE)
add_subdirectory("${LOCAL_SOURCE_ROOT}/whisper" whisper-build EXCLUDE_FROM_ALL)
add_library(trial_t5 SHARED IMPORTED)
set(t5_file "${T5_LIBRARY_ROOT}/${ANDROID_ABI}/libtrial_t5.so")
execute_process(COMMAND "${CMAKE_READELF}" --dynamic "${t5_file}"
    RESULT_VARIABLE audit_result OUTPUT_VARIABLE dynamic_info ERROR_QUIET)
if(NOT audit_result EQUAL 0 OR NOT dynamic_info MATCHES "\\(SONAME\\)[^\n]*\\[libtrial_t5\\.so\\]")
    message(FATAL_ERROR "Prepare the locked native runtime with its portable SONAME.")
endif()
set_target_properties(trial_t5 PROPERTIES IMPORTED_LOCATION "${t5_file}" IMPORTED_SONAME "libtrial_t5.so")
add_library(common_tongue_ai SHARED native.cpp)
target_include_directories(common_tongue_ai PRIVATE "${LOCAL_SOURCE_ROOT}/whisper/include")
target_link_options(common_tongue_ai PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(common_tongue_ai PRIVATE trial_t5 whisper android log)
