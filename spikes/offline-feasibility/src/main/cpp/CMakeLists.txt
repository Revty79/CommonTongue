cmake_minimum_required(VERSION 3.22)
project(offline_feasibility LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized native research inference in debug APK" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${SPIKE_SOURCE_ROOT}/whisper" whisper-build EXCLUDE_FROM_ALL)
add_subdirectory("${SPIKE_SOURCE_ROOT}/sentencepiece" sentencepiece-build EXCLUDE_FROM_ALL)
add_library(feasibility SHARED native.cpp)
target_include_directories(feasibility PRIVATE "${SPIKE_SOURCE_ROOT}/whisper/include" "${SPIKE_SOURCE_ROOT}/sentencepiece" "${SPIKE_SOURCE_ROOT}/sentencepiece/src")
target_link_libraries(feasibility PRIVATE whisper sentencepiece-static android log)
