cmake_minimum_required(VERSION 3.22)
project(physical_trial LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized research inference" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Prepared source dependencies only" FORCE)
# SentencePiece uses the legacy FetchContent_Populate signature. Point it at
# verified preparation explicitly; FULLY_DISCONNECTED alone does not stop it.
set(FETCHCONTENT_SOURCE_DIR_ABSEIL-CPP "${SPIKE_SOURCE_ROOT}/sentencepiece/third_party/abseil-cpp" CACHE PATH "Pinned Abseil" FORCE)
set(FETCHCONTENT_UPDATES_DISCONNECTED_ABSEIL-CPP ON CACHE BOOL "" FORCE)
add_subdirectory("${SPIKE_SOURCE_ROOT}/whisper" whisper-build EXCLUDE_FROM_ALL)
add_subdirectory("${SPIKE_SOURCE_ROOT}/sentencepiece" sentencepiece-build EXCLUDE_FROM_ALL)
add_library(trial_t5 SHARED IMPORTED)
set(trial_t5_file "${T5_LIBRARY_ROOT}/${ANDROID_ABI}/libtrial_t5.so")
# A missing SONAME makes the linker record this build-host path in DT_NEEDED.
# Reject a stale Rust library rather than producing an APK that cannot load.
execute_process(COMMAND "${CMAKE_READELF}" --dynamic "${trial_t5_file}"
    RESULT_VARIABLE trial_t5_readelf_result OUTPUT_VARIABLE trial_t5_dynamic ERROR_QUIET)
if(NOT trial_t5_readelf_result EQUAL 0 OR NOT trial_t5_dynamic MATCHES "\\(SONAME\\)[^\n]*\\[libtrial_t5\\.so\\]")
    message(FATAL_ERROR "Rebuild the pinned Rust runtime with SONAME libtrial_t5.so before building the research APK.")
endif()
set_target_properties(trial_t5 PROPERTIES IMPORTED_LOCATION "${trial_t5_file}" IMPORTED_SONAME "libtrial_t5.so")
add_library(device_trial SHARED native.cpp)
target_include_directories(device_trial PRIVATE "${SPIKE_SOURCE_ROOT}/whisper/include" "${SPIKE_SOURCE_ROOT}/sentencepiece" "${SPIKE_SOURCE_ROOT}/sentencepiece/src")
target_link_options(device_trial PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(device_trial PRIVATE trial_t5 whisper sentencepiece-static android log)
