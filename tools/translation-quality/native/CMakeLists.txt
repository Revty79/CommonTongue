cmake_minimum_required(VERSION 3.22)
project(common_tongue_quality_worker LANGUAGES C CXX)
if(NOT LLAMA_SOURCE)
    message(FATAL_ERROR "Set LLAMA_SOURCE to the pinned, locally prepared llama.cpp source")
endif()
file(TO_CMAKE_PATH "${LLAMA_SOURCE}" LLAMA_SOURCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama-runtime)
add_executable(quality-worker worker.cpp)
target_compile_features(quality-worker PRIVATE cxx_std_17)
target_include_directories(quality-worker PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(quality-worker PRIVATE llama)
add_executable(quality-official-simple "${LLAMA_SOURCE}/examples/simple/simple.cpp")
target_link_libraries(quality-official-simple PRIVATE llama)
if(MSVC)
    target_compile_options(quality-worker PRIVATE /utf-8)
endif()
